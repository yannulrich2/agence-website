/** * W.A.V.E. — Réception des leads du site vers Google Sheets * 1. Google Sheets > Extensions > Apps Script > colle ce code > Enregistrer * 2. Déployer > Nouveau déploiement > Type : Application Web * Exécuter en tant que : Moi | Qui a accès : Tout le monde * 3. Copie l'URL /exec et colle-la dans index.html (const SHEET_URL) */ const SHEET_NAME = 'Leads'; const NOTIFY_EMAIL = ''; // ex: 'toi@wavevisionary.com' pour recevoir un courriel à chaque lead (laisse vide pour désactiver) const COLUMNS = [ 'Date', 'Nom', 'Courriel', 'Téléphone', 'Site', 'Budget', 'Besoins', 'Message', 'utm_source', 'utm_medium', 'utm_campaign', 'utm_content', 'utm_term', 'fbclid', 'gclid', 'Page', 'Statut' ]; function doPost(e) { const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const p = e.parameter || {}; const sheet = getSheet_(); const row = [ new Date(), p.nom, p.courriel, p.telephone, p.site, p.budget, p.besoins, p.message, p.utm_source, p.utm_medium, p.utm_campaign, p.utm_content, p.utm_term, p.fbclid, p.gclid, p.page, 'Nouveau' ].map(v => (v === undefined ? '' : v)); sheet.appendRow(row); if (NOTIFY_EMAIL) { MailApp.sendEmail({ to: NOTIFY_EMAIL, subject: `Nouveau lead W.A.V.E. — ${p.nom || ''} (${p.budget || ''})`, body: `Nom : ${p.nom}\nCourriel : ${p.courriel}\nTéléphone : ${p.telephone}\nSite : ${p.site}\n` + `Budget : ${p.budget}\nBesoins : ${p.besoins}\nMessage : ${p.message}\n\n` + `Source : ${p.utm_source} / ${p.utm_medium} / ${p.utm_campaign}` }); } return ContentService.createTextOutput(JSON.stringify({ ok: true })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ ok: false, error: String(err) })) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } function getSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) sheet = ss.insertSheet(SHEET_NAME); if (sheet.getLastRow() === 0) { sheet.appendRow(COLUMNS); sheet.getRange(1, 1, 1, COLUMNS.length).setFontWeight('bold').setBackground('#0B0B0B').setFontColor('#FFFFFF'); sheet.setFrozenRows(1); } return sheet; } /** Lance cette fonction une fois depuis l'éditeur pour tester (ajoute une ligne test). */ function testLead() { doPost({ parameter: { nom: 'Test W.A.V.E.', courriel: 'test@test.com', site: 'test.com', budget: '5 000 – 20 000 $', besoins: 'Publicité', utm_source: 'test' } }); }
